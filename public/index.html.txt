                    <h3 class="text-md font-semibold text-slate-700 mb-2">Orodha ya Askari Waliopo (<span id="soldierCount">0</span>)</h3>

                        <!-- Orodha ya askari itaingia hapa -->

                <div class="bg-white p-5 rounded-xl shadow-md no-print flex flex-wrap gap-4 items-center justify-between">
                    <div>
                        <label class="block text-sm font-medium text-slate-600 mb-1">Chagua Mtindo wa Ratiba (Template):</label>

    <script>
        let soldiers = [];
        const posts = ["IKULU", "PBZ", "ULINZI KITUO", "CRO"];
        const shifts = ["Shift 1 (Saa 02:00 - 10:00)", "Shift 2 (Saa 10:00 - 18:00)", "Shift 3 (Saa 18:00 - 02:00)"];

        document.getElementById('currentDate').innerText = new Date().toLocaleDateString();

        document.getElementById('soldierForm').addEventListener('submit', function(e) {
            e.preventDefault();
            const name = document.getElementById('name').value;
            const serviceNumber = parseInt(document.getElementById('serviceNumber').value);

            soldiers.push({ name, serviceNumber });
            // Panga askari kuanzia namba ndogo kwenda kubwa au kinyume chake
            soldiers.sort((a, b) => a.serviceNumber - b.serviceNumber);

            document.getElementById('name').value = '';
            document.getElementById('serviceNumber').value = '';
            updateSoldiersUI();
        });

        function updateSoldiersUI() {
            const listContainer = document.getElementById('soldiersList');
            document.getElementById('soldierCount').innerText = soldiers.length;
            listContainer.innerHTML = '';

            soldiers.forEach((s, index) => {
                listContainer.innerHTML += `
                    <div class="flex justify-between items-center bg-white p-2 rounded border text-sm">
                        <span><b>${s.name}</b> (No: ${s.serviceNumber})</span>
                        <button onclick="removeSoldier(${index})" class="text-red-500 hover:text-red-700 font-bold px-2">×</button>
                    </div>
                `;
            });
        }

        function removeSoldier(index) {
            soldiers.splice(index, 1);
            updateSoldiersUI();
        }

        function generateSchedule() {
            // Malindo 4 * Shift 3 = Shift 12 zinahitajika. Kila shift ina Senior 1 na Junior 1 (Jumla askari 24 wanahitajika)
            if (soldiers.length < 24) {
                alert("Tafadhali sajili angalau askari 24 ili kukamilisha malindo yote 4 na shift zake tatu.");
                return;
            }

            let availableSoldiers = [...soldiers];
            let htmlContent = '';
            const template = document.getElementById('templateSelector').value;

            posts.forEach(post => {
                let postHtml = '';
                let shiftData = [];

                shifts.forEach(shift => {
                    let senior = availableSoldiers.shift();
                    let junior = availableSoldiers.shift();

                    if (!senior) senior = { name: "Nafasi Wazi", serviceNumber: "-" };
                    if (!junior) junior = { name: "Nafasi Wazi", serviceNumber: "-" };

                    shiftData.push({ shift, senior, junior });
                });

                if (template === 'modern') {
                    postHtml = `
                        <div class="border border-indigo-100 rounded-lg p-4 bg-indigo-50/30 mb-4">
                            <h3 class="font-bold text-indigo-900 text-base mb-3 border-b border-indigo-200 pb-1">📍 LINDO: ${post}</h3>
                            <div class="grid grid-cols-1 md:grid-cols-3 gap-3">
                    `;
                    shiftData.forEach(d => {
                        postHtml += `
                            <div class="bg-white p-3 rounded border shadow-sm">
                                <div class="text-xs font-bold text-indigo-600 mb-1">${d.shift}</div>
                                <div class="text-sm">⭐ <b>Senior:</b> ${d.senior.name} (${d.senior.serviceNumber})</div>
                                <div class="text-sm">🛡️ <b>Junior:</b> ${d.junior.name} (${d.junior.serviceNumber})</div>
                            </div>
                        `;
                    });
                    postHtml += `</div></div>`;
                } else {
                    postHtml = `
                        <div class="mb-4">
                            <h3 class="font-bold text-slate-800 bg-slate-200 p-2 text-sm">📍 ${post}</h3>
                            <table class="w-full text-left border-collapse text-sm">
                                <thead>
                                    <tr class="bg-slate-100 border-b">
                                        <th class="p-2 border">Shift</th>
                                        <th class="p-2 border">Senior Guard</th>
                                        <th class="p-2 border">Junior Guard</th>
                                    </tr>
                                </thead>
                                <tbody>
                    `;
                    shiftData.forEach(d => {
                        postHtml += `
                            <tr class="border-b">
                                <td class="p-2 border font-medium text-slate-600">${d.shift}</td>
                                <td class="p-2 border">⭐ ${d.senior.name} (${d.senior.serviceNumber})</td>
                                <td class="p-2 border">🛡️ ${d.junior.name} (${d.junior.serviceNumber})</td>
                            </tr>
                        `;
                    });
                    postHtml += `</tbody></table></div>`;
                }

                htmlContent += postHtml;
            });

            document.getElementById('scheduleContent').innerHTML = htmlContent;
        }

        function changeTemplate() {
            if(document.getElementById('scheduleContent').innerHTML.trim() !== '') {
                generateSchedule();
            }
        }
    </script>
